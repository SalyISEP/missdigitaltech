import { inject } from '@angular/core';
import { CanActivateFn, Router } from '@angular/router';
import { AuthService } from '../services/auth.service';

export const adminGuard: CanActivateFn = () => {
  const auth = inject(AuthService);
  const router = inject(Router);

  if (!auth.isLoggedIn()) {
    return router.createUrlTree(['/login']);
  }

  if (auth.isAdmin()) {
    return true;
  }

  return router.createUrlTree(['/espace-client/dashboard']);
};

===== FILE: src\app\core\interceptors\auth.interceptor.ts =====
import { HttpInterceptorFn } from '@angular/common/http';

export const authInterceptor: HttpInterceptorFn = (req, next) => {

  const token = localStorage.getItem('mdt_access_token');

  if (!token) {
    return next(req);
  }

  const authReq = req.clone({
    setHeaders: {
      Authorization: `Bearer ${token}`
    }
  });

  return next(authReq);
};

import { Injectable } from '@angular/core';
import { HttpClient, HttpHeaders } from '@angular/common/http';
import { Observable } from 'rxjs';
import { environment } from '../../../environments/environment';

@Injectable({
  providedIn: 'root'
})
export class ApiService {
  private readonly baseUrl = environment.apiUrl;

  constructor(private http: HttpClient) {}

  get<T>(endpoint: string): Observable<T> {
    return this.http.get<T>(
      `${this.baseUrl}/${endpoint}`,
      this.options()
    );
  }

  post<T>(endpoint: string, data: unknown): Observable<T> {
    return this.http.post<T>(
      `${this.baseUrl}/${endpoint}`,
      data,
      this.options()
    );
  }

  put<T>(endpoint: string, data: unknown): Observable<T> {
    return this.http.put<T>(
      `${this.baseUrl}/${endpoint}`,
      data,
      this.options()
    );
  }

  delete<T>(endpoint: string): Observable<T> {
    return this.http.delete<T>(
      `${this.baseUrl}/${endpoint}`,
      this.options()
    );
  }

  private options() {
    const token = localStorage.getItem('mdt_access_token');

    let headers = new HttpHeaders({
      'Content-Type': 'application/json'
    });

    if (token) {
      headers = headers.set(
        'Authorization',
        `Bearer ${token}`
      );
    }

    return { headers };
  }
}

import { Injectable } from '@angular/core';
import { Observable, tap } from 'rxjs';
import { ApiService } from './api.service';

interface LoginResponse {
  access: string;
  refresh: string;
  user?: unknown;
}

@Injectable({
  providedIn: 'root'
})
export class AuthApiService {
  constructor(private api: ApiService) {}

  login(
    email: string,
    password: string
  ): Observable<LoginResponse> {
    return this.api
      .post<LoginResponse>('auth/login/', {
        email,
        password
      })
      .pipe(
        tap(response => {
          localStorage.setItem(
            'mdt_access_token',
            response.access
          );

          localStorage.setItem(
            'mdt_refresh_token',
            response.refresh
          );
        })
      );
  }

  register(data: unknown): Observable<unknown> {
    return this.api.post(
      'auth/register/',
      data
    );
  }

  refresh(): Observable<{ access: string }> {
    const refresh =
      localStorage.getItem('mdt_refresh_token');

    return this.api
      .post<{ access: string }>(
        'auth/token/refresh/',
        { refresh }
      )
      .pipe(
        tap(response => {
          localStorage.setItem(
            'mdt_access_token',
            response.access
          );
        })
      );
  }

  logout(): void {
    localStorage.removeItem(
      'mdt_access_token'
    );

    localStorage.removeItem(
      'mdt_refresh_token'
    );

import { Injectable } from '@angular/core';
import { HttpClient } from '@angular/common/http';
import { Observable } from 'rxjs';
import { map, switchMap, tap } from 'rxjs/operators';
import { User } from '../models/user.model';
import { environment } from '../../../environments/environment';

interface JwtResponse {
  access: string;
  refresh: string;
}

interface ApiUser {
  id: number | string;
  username?: string;
  first_name?: string;
  last_name?: string;
  email: string;
  phone?: string;
  role?: string;
}

export interface RegisterData {
  email: string;
  password: string;
  firstName?: string;
  lastName?: string;
  first_name?: string;
  last_name?: string;
  phone?: string;
}

@Injectable({
  providedIn: 'root'
})
export class AuthService {

  private readonly api = environment.apiUrl;
  private readonly accessKey = 'mdt_access_token';
  private readonly refreshKey = 'mdt_refresh_token';
  private readonly userKey = 'mdt_user';

  private currentUser: User | null = this.loadUser();

  constructor(private http: HttpClient) {}

  login(credentials: {
    email: string;
    password: string;
  }): Observable<User> {
    return this.http
      .post<JwtResponse>(`${this.api}/auth/login/`, credentials)
      .pipe(
        tap(tokens => {
          localStorage.setItem(this.accessKey, tokens.access);
          localStorage.setItem(this.refreshKey, tokens.refresh);
        }),
        switchMap(() => this.loadCurrentUser())
      );
  }

  register(data: RegisterData): Observable<any> {
    const payload = {
      email: data.email,
      password: data.password,
      first_name: data.first_name || data.firstName || '',
      last_name: data.last_name || data.lastName || '',
      phone: data.phone || ''
    };

    return this.http.post(`${this.api}/auth/register/`, payload);
  }

  loadCurrentUser(): Observable<User> {
    return this.http
      .get<ApiUser>(`${this.api}/users/me/`)
      .pipe(
        map(user => this.mapUser(user)),
        tap(user => {
          this.currentUser = user;
          localStorage.setItem(this.userKey, JSON.stringify(user));
        })
      );
  }

  getCurrentUser(): User | null {
    return this.currentUser;
  }

  getUser(): User | null {
    return this.currentUser;
  }

  requestPasswordReset(email: string): Observable<any> {
    return this.http.post(`${this.api}/auth/password-reset/`, { email });
  }

  getToken(): string | null {
    return localStorage.getItem(this.accessKey);
  }

  isLoggedIn(): boolean {
    return !!this.getToken();
  }

  isAdmin(): boolean {
    return this.currentUser?.role === 'admin';
  }

  logout(): void {
    localStorage.removeItem(this.accessKey);
    localStorage.removeItem(this.refreshKey);
    localStorage.removeItem(this.userKey);
    this.currentUser = null;
  }

  private mapUser(user: ApiUser): User {
    return {
      id: String(user.id),
      firstName: user.first_name || '',
      lastName: user.last_name || '',
      email: user.email,
      phone: user.phone || '',
      role: user.role === 'admin' ? 'admin' : 'client'
    };
  }

  private loadUser(): User | null {
    const stored = localStorage.getItem(this.userKey);

    if (!stored) {
      return null;
    }

    try {
      return JSON.parse(stored) as User;
    } catch {
      localStorage.removeItem(this.userKey);
      return null;
    }
  }
}

import { Component, OnInit } from '@angular/core';
import { CommonModule } from '@angular/common';
import { FormsModule } from '@angular/forms';
import { ActivatedRoute, RouterLink } from '@angular/router';
import { Formation, FormationService } from '../../../../core/services/formation.service';
import { ReservationService } from '../../../../core/services/reservation.service';
import { PaymentService } from '../../../../core/services/payment.service';
import { AuthService } from '../../../../core/services/auth.service';

@Component({
  selector: 'app-formation-detail',
  standalone: true,
  imports: [CommonModule, FormsModule, RouterLink],
  templateUrl: './formation-detail.component.html',
  styleUrl: './formation-detail.component.scss'
})
export class FormationDetailComponent implements OnInit {
  formation: Formation | null = null;
  loading = true;
  error = '';
  success = '';
  date = '';
  time = '';
  notes = '';
  submitting = false;

  reservationId: number | null = null;
  paymentMethod: 'wave' | 'orange_money' = 'wave';
  paymentSubmitting = false;
  paymentMessage = '';
  paymentError = '';

  constructor(
    private route: ActivatedRoute,
    private formationService: FormationService,
    private reservationService: ReservationService,
    private paymentService: PaymentService,
    public authService: AuthService
  ) {}

  ngOnInit(): void {
    const id = Number(this.route.snapshot.paramMap.get('id'));
    if (!id) {
      this.error = 'Formation introuvable.';
      this.loading = false;
      return;
    }

    this.formationService.getFormation(id).subscribe({
      next: formation => {
        this.formation = formation;
        this.loading = false;
      },
      error: () => {
        this.error = 'Impossible de charger cette formation.';
        this.loading = false;
      }
    });
  }

  reserver(): void {
    if (!this.formation || !this.date || this.submitting) return;

    this.submitting = true;
    this.error = '';
    this.success = '';
    this.paymentMessage = '';
    this.paymentError = '';
    this.reservationId = null;

    this.reservationService.createReservation({
      formation: this.formation.id,
      date: this.date,
      ...(this.time ? { time: this.time } : {}),
      notes: this.notes
    }).subscribe({
      next: reservation => {
        this.reservationId = reservation.id;
        this.success = 'RÃ©servation enregistrÃ©e. Vous pouvez maintenant prÃ©parer votre paiement.';
        this.submitting = false;
        this.date = '';
        this.time = '';
        this.notes = '';
      },
      error: err => {
        console.error('Erreur rÃ©servation :', err);
        const detail = err?.error;
        this.error = detail?.date?.[0] ||
          detail?.time?.[0] ||
          detail?.formation?.[0] ||
          detail?.detail ||
          'Impossible dâ€™enregistrer la rÃ©servation. VÃ©rifiez les champs saisis.';
        this.submitting = false;
      }
    });
  }

  preparerPaiement(): void {
    const reservationId = this.reservationId;
    if (reservationId === null || this.paymentSubmitting) return;

    this.paymentSubmitting = true;
    this.paymentError = '';
    this.paymentMessage = '';

    this.paymentService.createPayment({
      reservation: reservationId,
      method: this.paymentMethod
    }).subscribe({
      next: () => {
        this.paymentMessage =
          'Demande de paiement enregistrÃ©e en attente. Le paiement en ligne sera activÃ© aprÃ¨s la configuration du compte marchand.';

          err?.error?.reservation?.[0] ||
          err?.error?.method?.[0] ||
          'Impossible de crÃ©er la demande de paiement. RÃ©essayez.';
        this.paymentSubmitting = false;
      }
    });
  }
}

  selector: 'app-navbar',
  standalone: true,
  imports: [RouterLink],
  templateUrl: './navbar.component.html',
  styleUrl: './navbar.component.scss'
})
export class NavbarComponent {
  constructor(
    public auth: AuthService,
    private router: Router
  ) {}

  logout(): void {
    this.auth.logout();
  }

  goToDashboard(): void {
    if (this.auth.isAdmin()) {
      this.router.navigate(['/admin/dashboard']);
    } else {
      this.router.navigate(['/dashboard']);
    }
  }
}
